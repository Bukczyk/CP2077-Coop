cmake_minimum_required(VERSION 3.21)
project(CP2077CoopSession VERSION 0.1.0 LANGUAGES CXX)

option(COOP_BUILD_PLUGIN "Build the Windows RED4ext legacy plugin" OFF)
option(COOP_BUILD_LEGACY_SERVER "Build the frozen Windows relay" ${WIN32})
option(COOP_SANITIZE "Enable Address/Undefined sanitizers on GCC/Clang" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_subdirectory(shared)
if(COOP_BUILD_PLUGIN)
    if(NOT WIN32)
        message(FATAL_ERROR "The RED4ext plugin requires Windows")
    endif()
    add_subdirectory(CoopPlugin)
endif()
if(COOP_BUILD_LEGACY_SERVER)
    if(NOT WIN32)
        message(FATAL_ERROR "Legacy relay requires Windows; the portable session core is built separately")
    endif()
    add_subdirectory(CoopServer)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
