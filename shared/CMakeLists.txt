add_library(coop_core STATIC src/protocol.cpp src/session.cpp src/net.cpp src/interpolation.cpp src/server.cpp src/client.cpp src/game_bridge.cpp)
add_library(coop::core ALIAS coop_core)
target_include_directories(coop_core PUBLIC include)
target_compile_features(coop_core PUBLIC cxx_std_20)
if(BUILD_TESTING)
    target_compile_definitions(coop_core PUBLIC COOP_TESTING=1)
endif()
if(MSVC)
    target_compile_options(coop_core PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(coop_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(COOP_SANITIZE)
        target_compile_options(coop_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(coop_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()

if(WIN32)
    target_link_libraries(coop_core PUBLIC ws2_32 bcrypt)
endif()

find_package(Threads REQUIRED)
target_link_libraries(coop_core PUBLIC Threads::Threads)
