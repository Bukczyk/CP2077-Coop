cmake_minimum_required(VERSION 3.21)
project(CP2077Coop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
    message(FATAL_ERROR "CP2077Coop plugin requires Windows")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../build-support/RED4ext.cmake")
option(COOP_LEGACY_PLUGIN "Build frozen rollback plugin instead of typed session bridge" OFF)
if(COOP_LEGACY_PLUGIN)
    add_library(CP2077Coop SHARED src/legacy_main.cpp)
    target_link_libraries(CP2077Coop PRIVATE RED4ext::SDK ws2_32)
else()
    if(NOT TARGET coop_core)
        add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../shared" "${CMAKE_CURRENT_BINARY_DIR}/shared")
    endif()
    add_library(CP2077Coop SHARED src/main.cpp)
    target_link_libraries(CP2077Coop PRIVATE RED4ext::SDK coop::core)
endif()
